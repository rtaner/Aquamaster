#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include "RTClib.h"
#include <ArduinoJson.h>
#include "time.h"
#include <WebServer.h>

// --- GÜVENLİK BİLGİLERİ ---
const char* ssid = "YOUR_WIFI_SSID"; 
const char* password = "YOUR_WIFI_PASSWORD";
String supabaseUrl = "YOUR_SUPABASE_URL"; 
String supabaseKey = "YOUR_SUPABASE_KEY"; 
// ------------------------------------

const char* ntpServer = "pool.ntp.org";
const long  gmtOffset_sec = 3 * 3600; 
const int   daylightOffset_sec = 0;   

RTC_DS3231 rtc;
WebServer server(80); // Lokal Anlık Bas-Tut Hortum Doldurma için WebSunucusu

const int POMPA_1 = 16;
const int POMPA_2 = 17;
const int POMPA_3 = 18;
const int POMPA_4 = 19;

// Hortum Havasını Alma (Priming) Güvenlik ve Canlılık Takibi (Watchdog)
unsigned long primeLastHeartbeat[5] = {0, 0, 0, 0, 0};
bool pumpPrimingActive[5] = {false, false, false, false, false};

// --- OFFLINE HAFIZA (CACHE) YAPISI ---
struct HafizaGorev {
  int pump_id;
  int hour;
  int minute;
  int duration;
  char schedule_type[12]; // "daily", "weekly", "interval"
  bool days_of_week[8];   // Index 1..7 (1: Pzt, 2: Sal, ..., 7: Pzr)
  int interval_days;
  uint32_t start_timestamp; // UNIX timestamp for interval calculation
};
HafizaGorev gunlukProgramlar[20]; // Maksimum 20 görev hafızada tutulabilir
int hafizadakiGorevSayisi = 0;

// Her pompanın aynı dakika içinde birden fazla çalışmasını engelleyen bağımsız sayaçlar (0 kullanılmaz, 1-2-3-4 için)
int sonCalisanDakika[5] = {-1, -1, -1, -1, -1}; 

unsigned long sonİnternetKontrol = 0;
unsigned long sonGorevKontrol = 0;

void handlePrime() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  if (!server.hasArg("pump") || !server.hasArg("state")) {
    server.send(400, "text/plain", "Eksik parametre");
    return;
  }

  int pump = server.arg("pump").toInt();
  String state = server.arg("state");

  int pin = -1;
  if (pump == 1) pin = POMPA_1;
  else if (pump == 2) pin = POMPA_2;
  else if (pump == 3) pin = POMPA_3;
  else if (pump == 4) pin = POMPA_4;

  if (pin != -1) {
    if (state == "on") {
      digitalWrite(pin, HIGH);
      pumpPrimingActive[pump] = true;
      primeLastHeartbeat[pump] = millis();
      server.send(200, "text/plain", "PRIMING_ON");
    } else {
      digitalWrite(pin, LOW);
      pumpPrimingActive[pump] = false;
      server.send(200, "text/plain", "PRIMING_OFF");
    }
  } else {
    server.send(400, "text/plain", "Gecersiz pompa");
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(POMPA_1, OUTPUT); pinMode(POMPA_2, OUTPUT); pinMode(POMPA_3, OUTPUT); pinMode(POMPA_4, OUTPUT);
  digitalWrite(POMPA_1, LOW); digitalWrite(POMPA_2, LOW); digitalWrite(POMPA_3, LOW); digitalWrite(POMPA_4, LOW);

  if (!rtc.begin()) {
    Serial.println("RTC modulu bulunamadi!");
    while (1); 
  }

  Serial.print("\nWiFi'ye baglaniliyor: ");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Baglantisi Tamam!");
  Serial.print("Lokal IP Adresi: "); Serial.println(WiFi.localIP());

  server.on("/prime", handlePrime);
  server.begin();
  Serial.println("Lokal WebServer Baslatildi (Port 80)");

  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
  struct tm timeinfo;
  if (getLocalTime(&timeinfo)) {
    rtc.adjust(DateTime(timeinfo.tm_year + 1900, timeinfo.tm_mon + 1, timeinfo.tm_mday, timeinfo.tm_hour, timeinfo.tm_min, timeinfo.tm_sec));
    Serial.println(">>> BASARILI: Saat guncellendi! <<<");
  }
}

void goreviPasifYap(int gorevID) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    String endpoint = supabaseUrl + "/rest/v1/schedules?id=eq." + String(gorevID);
    http.begin(endpoint);
    http.addHeader("apikey", supabaseKey);
    http.addHeader("Authorization", "Bearer " + supabaseKey);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Prefer", "return=minimal");
    http.PATCH("{\"is_active\": false}");
    http.end();
  }
}

void motorCalistir(int motorNo, int saniye) {
  int hedefPin = -1;
  if (motorNo == 1) hedefPin = POMPA_1;
  else if (motorNo == 2) hedefPin = POMPA_2;
  else if (motorNo == 3) hedefPin = POMPA_3;
  else if (motorNo == 4) hedefPin = POMPA_4;

  if (hedefPin != -1) {
    Serial.print("\n>>> "); 
    Serial.print(motorNo); Serial.print(". POMPA "); 
    Serial.print(saniye); Serial.println(" SN CALISIYOR... <<<");
    
    digitalWrite(hedefPin, HIGH); 
    delay(saniye * 1000);         
    digitalWrite(hedefPin, LOW);  
    
    Serial.println(">>> ISLEM TAMAM! <<<\n");
  }
}

void logEkle(int pumpId, int durationSeconds) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    String endpoint = supabaseUrl + "/rest/v1/dosing_logs";
    http.begin(endpoint);
    http.addHeader("apikey", supabaseKey);
    http.addHeader("Authorization", "Bearer " + supabaseKey);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Prefer", "return=minimal");
    String body = "{\"pump_id\":" + String(pumpId) + ",\"duration_seconds\":" + String(durationSeconds) + ",\"ml_amount\":" + String(durationSeconds) + ",\"mode\":\"Zamanlanmış\"}";
    http.POST(body);
    http.end();
  }
}

void kalpatisiGonder() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    String endpoint = supabaseUrl + "/rest/v1/device_status?id=eq.1";
    http.begin(endpoint);
    http.addHeader("apikey", supabaseKey);
    http.addHeader("Authorization", "Bearer " + supabaseKey);
    http.addHeader("Content-Type", "application/json");
    String localIP = WiFi.localIP().toString();
    String body = "{\"last_seen\":\"now()\",\"ip_address\":\"" + localIP + "\"}";
    int httpCode = http.PATCH(body);
    
    // Eğer Supabase'de ip_address sütunu yoksa ve PATCH 400 hatası verirse, yedek olarak sadece last_seen gönder!
    if (httpCode != 200 && httpCode != 204 && httpCode != 201) {
      HTTPClient http2;
      http2.begin(endpoint);
      http2.addHeader("apikey", supabaseKey);
      http2.addHeader("Authorization", "Bearer " + supabaseKey);
      http2.addHeader("Content-Type", "application/json");
      http2.PATCH("{\"last_seen\":\"now()\"}");
      http2.end();
    }
    http.end();
  }
}

void loop() {
  DateTime now = rtc.now(); 
  server.handleClient(); // Lokal HTTP İsteklerini (Bas-Tut Doldurma) Dinle

  // --- GÜVENLİK KİLİDİ (WATCHDOG TIMER) ---
  // Hortum doldurma sırasında Wi-Fi koparsa veya 1.5 sn sinyal gelmezse motoru anında kapat!
  for (int p = 1; p <= 4; p++) {
    if (pumpPrimingActive[p]) {
      if (millis() - primeLastHeartbeat[p] > 1500) {
        int pin = (p == 1) ? POMPA_1 : (p == 2) ? POMPA_2 : (p == 3) ? POMPA_3 : POMPA_4;
        digitalWrite(pin, LOW);
        pumpPrimingActive[p] = false;
        Serial.print("[GUVENLIK] Sinyal koptugu icin "); Serial.print(p); Serial.println(". Pompa kapatildi!");
      }
    }
  } 

  // --- 1. AŞAMA: İNTERNET SENKRONİZASYONU VE KALP ATIŞI (Her 10 Saniyede Bir) ---
  if (millis() - sonİnternetKontrol > 10000) {
    sonİnternetKontrol = millis();
    
    if (WiFi.status() == WL_CONNECTED) {
      kalpatisiGonder(); // Canlılık sinyalini Supabase'e gönder

      HTTPClient http;
      String endpoint = supabaseUrl + "/rest/v1/schedules?select=*&is_active=eq.true";
      http.begin(endpoint);
      http.addHeader("apikey", supabaseKey);
      http.addHeader("Authorization", "Bearer " + supabaseKey);
      
      int httpResponseCode = http.GET();
      if (httpResponseCode == 200) {
        String payload = http.getString();
        DynamicJsonDocument doc(4096);
        DeserializationError error = deserializeJson(doc, payload);
        
        if (!error) {
          int geciciGorevSayisi = 0;
          HafizaGorev geciciHafiza[20];

          for (JsonObject item : doc.as<JsonArray>()) {
            int id = item["id"];
            int pump_id = item["pump_id"];
            int duration = item["duration_seconds"];
            bool is_one_time = item["is_one_time"];
            
            // Eğer manuel anlık tetikleme ise anında yap ve veritabanından sil!
            if (is_one_time == true) {
              Serial.println("\n[ANLIK GOREV] Arayuzden manuel tetikleme geldi!");
              motorCalistir(pump_id, duration);
              goreviPasifYap(id); 
            } 
            // Eğer programlı görev ise, saati geldiğinde yapmak üzere HAFIZAYA AL
            else {
              String run_time = item["run_time"].as<String>(); 
              geciciHafiza[geciciGorevSayisi].pump_id = pump_id;
              geciciHafiza[geciciGorevSayisi].hour = run_time.substring(0, 2).toInt();
              geciciHafiza[geciciGorevSayisi].minute = run_time.substring(3, 5).toInt();
              geciciHafiza[geciciGorevSayisi].duration = duration;

              // Esnek Zamanlama Alanları
              String stype = item["schedule_type"] | "daily";
              strncpy(geciciHafiza[geciciGorevSayisi].schedule_type, stype.c_str(), sizeof(geciciHafiza[geciciGorevSayisi].schedule_type));

              for(int d = 1; d <= 7; d++) geciciHafiza[geciciGorevSayisi].days_of_week[d] = false;
              if (item.containsKey("days_of_week") && !item["days_of_week"].isNull()) {
                JsonArray days = item["days_of_week"];
                for(int d : days) {
                  if(d >= 1 && d <= 7) geciciHafiza[geciciGorevSayisi].days_of_week[d] = true;
                }
              } else {
                for(int d = 1; d <= 7; d++) geciciHafiza[geciciGorevSayisi].days_of_week[d] = true;
              }

              geciciHafiza[geciciGorevSayisi].interval_days = item["interval_days"] | 1;

              if (item.containsKey("start_date") && !item["start_date"].isNull()) {
                String sDate = item["start_date"].as<String>();
                int yr = sDate.substring(0, 4).toInt();
                int mo = sDate.substring(5, 7).toInt();
                int dy = sDate.substring(8, 10).toInt();
                DateTime dt(yr, mo, dy, 0, 0, 0);
                geciciHafiza[geciciGorevSayisi].start_timestamp = dt.unixtime();
              } else {
                geciciHafiza[geciciGorevSayisi].start_timestamp = now.unixtime();
              }

              geciciGorevSayisi++;
            }
          }
          
          // İnternetten çekilen programlı görevleri ESP32'nin ana hafızasına aktar
          hafizadakiGorevSayisi = geciciGorevSayisi;
          for(int i = 0; i < geciciGorevSayisi; i++) {
            gunlukProgramlar[i] = geciciHafiza[i];
          }
        }
      }
      http.end();
    } else {
      Serial.println("[UYARI] WiFi Baglantisi Koptu! Offline (Cevrimdisi) mod devrede.");
    }
  }

  // --- 2. AŞAMA: OFFLINE HAFIZA KONTROLÜ VE MOTOR TETİKLEME (Her Saniye) ---
  if (millis() - sonGorevKontrol > 1000) {
    sonGorevKontrol = millis();
    
    // ESP32 kendi hafızasındaki listeyi sürekli kontrol eder (İnternet olsa da olmasa da)
    for(int i = 0; i < hafizadakiGorevSayisi; i++) {
      int pId = gunlukProgramlar[i].pump_id;
      
      if(now.hour() == gunlukProgramlar[i].hour && now.minute() == gunlukProgramlar[i].minute) {
        // Bu motor bu dakika içinde henüz çalışmadıysa zamanlama kuralını denetle!
        if(sonCalisanDakika[pId] != now.minute()) {
          bool calismaliMi = false;
          String stype = String(gunlukProgramlar[i].schedule_type);

          if (stype == "weekly") {
            int bugunNo = now.dayOfTheWeek();
            if (bugunNo == 0) bugunNo = 7; // Pazar = 7 (1: Pzt ... 7: Pzr)
            if (gunlukProgramlar[i].days_of_week[bugunNo]) {
              calismaliMi = true;
            }
          } else if (stype == "interval") {
            int interval = gunlukProgramlar[i].interval_days;
            if (interval <= 1) {
              calismaliMi = true;
            } else {
              uint32_t simdiUnix = now.unixtime();
              uint32_t baslangicUnix = gunlukProgramlar[i].start_timestamp;
              if (simdiUnix >= baslangicUnix) {
                long gunFarki = (simdiUnix - baslangicUnix) / 86400;
                if (gunFarki % interval == 0) {
                  calismaliMi = true;
                }
              }
            }
          } else {
            // "daily"
            calismaliMi = true;
          }

          if (calismaliMi) {
            Serial.println("\n[OTOMATIK PROGRAM] Hafizadaki gorev calistiriliyor!");
            motorCalistir(pId, gunlukProgramlar[i].duration);
            logEkle(pId, gunlukProgramlar[i].duration); // Otomatik çalıştırma log kaydını Supabase'e ekle
            sonCalisanDakika[pId] = now.minute();
          }
        }
      }
    }
  }
}

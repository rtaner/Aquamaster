// --- GÜVENLİ VE ÇAKIŞMASIZ DS18B20 SICAKLIK OKUMA FONKSİYONU ---
float sicaklikOku() {
  sensors.requestTemperatures();
  float temp = sensors.getTempCByIndex(0);

  // Okuma anlık zamanlamaya takılırsa 150ms bekleyip 1 kez daha dene (ESP32 Timing Fix)
  if (temp == DEVICE_DISCONNECTED_C || temp <= -50.0) {
    delay(150);
    sensors.requestTemperatures();
    temp = sensors.getTempCByIndex(0);
  }

  // Geçerli okuma alındıysa global değişkeni güncelle, alınamadıysa son geçerli değeri koru
  if (temp != DEVICE_DISCONNECTED_C && temp > -50.0) {
    suSicakligi = temp;
  }
  
  return suSicakligi;
}

// --- TDS METER V1.0 HASSAS ANALOG VE SICAKLIK KOMPANZASYONLU OKUMA FONKSİYONU ---
void tdsEcOku(float &tdsPpmResult, float &ecValResult) {
  int analogBuffer[30];
  int analogBufferTemp[30];
  
  // 30 Örnek al (300 ms)
  for (int i = 0; i < 30; i++) {
    analogBuffer[i] = analogRead(TDS_PIN);
    delay(10);
  }
  
  // Sıralama (Bubble Sort - Medyan Filtresi için)
  for (int i = 0; i < 30; i++) {
    analogBufferTemp[i] = analogBuffer[i];
  }
  for (int i = 0; i < 29; i++) {
    for (int j = i + 1; j < 30; j++) {
      if (analogBufferTemp[i] > analogBufferTemp[j]) {
        int temp = analogBufferTemp[i];
        analogBufferTemp[i] = analogBufferTemp[j];
        analogBufferTemp[j] = temp;
      }
    }
  }
  
  // Medyan 10 örneğin ortalamasını al
  long medianSum = 0;
  for (int i = 10; i < 20; i++) {
    medianSum += analogBufferTemp[i];
  }
  float averageAdc = (float)medianSum / 10.0;
  
  // Voltaj Dönüşümü (ESP32 ADC 12-bit: 0..4095, 3.3V Referans)
  float rawVoltage = averageAdc * (3.3 / 4095.0);
  
  // Anlık DS18B20 sıcaklığı ile katsayı hesaplama (25°C referansı)
  float mevcutSicaklik = (suSicakligi > -50.0) ? suSicakligi : 25.0;
  float tempCoefficient = 1.0 + 0.02 * (mevcutSicaklik - 25.0);
  float compensatedVoltage = rawVoltage / tempCoefficient;
  
  // TDS (PPM) Dönüşümü
  float tdsVal = (133.42 * pow(compensatedVoltage, 3) - 255.86 * pow(compensatedVoltage, 2) + 857.39 * compensatedVoltage) * 0.5;
  if (tdsVal < 0) tdsVal = 0;
  
  // EC (uS/cm) Dönüşümü
  float ecVal = tdsVal / 0.5;
  
  suTdsPpm = tdsVal;
  suEcVal = ecVal;
  tdsPpmResult = tdsVal;
  ecValResult = ecVal;
}


// --- LOKAL ANLIK TDS/EC ÖLÇÜM (KAYITSIZ TEST) ROUTE ---
void handleReadTds() {

}

// --- LOKAL HORTUM ÇEKME (PRIMING) ROUTE ---
void handlePrime() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  if (!server.hasArg("pump") || !server.hasArg("state")) {
    server.send(400, "text/plain", "Eksik parametre");
    return;
  }

  int pump = server.arg("pump").toInt();
  String state = server.arg("state");

  int pin = -1;
  if (pump == 1) pin = POMPA_1;
  else if (pump == 2) pin = POMPA_2;
  else if (pump == 3) pin = POMPA_3;
  else if (pump == 4) pin = POMPA_4;

  if (pin != -1) {
    if (state == "on") {
      digitalWrite(pin, HIGH);
      pumpPrimingActive[pump] = true;
      primeLastHeartbeat[pump] = millis();
      server.send(200, "text/plain", "PRIMING_ON");
    } else {
      digitalWrite(pin, LOW);
      pumpPrimingActive[pump] = false;
      server.send(200, "text/plain", "PRIMING_OFF");
    }
  } else {
    server.send(400, "text/plain", "Gecersiz pompa");
  }
}

// --- GLOBAL ACİL DURDURMA (E-STOP) ROUTE ---
void handleStop() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  
  // Tüm motor pinlerini anında LOW (kapat) yap!
  digitalWrite(POMPA_1, LOW);
  digitalWrite(POMPA_2, LOW);
  digitalWrite(POMPA_3, LOW);
  digitalWrite(POMPA_4, LOW);

  for (int p = 1; p <= 4; p++) {
    pumpPrimingActive[p] = false;
  }
  
  emergencyStopActive = true; // Çalışan motor loop'unu anında kıran bayrak
  server.send(200, "text/plain", "STOP_OK");
  Serial.println("\n[ACIL DURDUR] Arayuzden acil durdurma emri geldi! Tum pompalar aninda kesildi!");
}

// --- NON-BLOCKING & ANLIK ACİL DURDURMA DESTEKLİ HASSAS MİLİSANİYE MOTOR ÇALIŞTIRMA ---
void motorCalistir(int motorNo, unsigned long sureMs) {
  int hedefPin = -1;
  if (motorNo == 1) hedefPin = POMPA_1;
  else if (motorNo == 2) hedefPin = POMPA_2;
  else if (motorNo == 3) hedefPin = POMPA_3;
  else if (motorNo == 4) hedefPin = POMPA_4;

  if (hedefPin != -1 && sureMs > 0) {
    emergencyStopActive = false; // Yeni motor başlangıcında bayrağı sıfırla
    Serial.print("\n>>> "); 
    Serial.print(motorNo); Serial.print(". POMPA "); 
    Serial.print(sureMs); Serial.print(" MS (");
    Serial.print((float)sureMs / 1000.0, 2); Serial.println(" SN) CALISIYOR... <<<");
    
    digitalWrite(hedefPin, HIGH); 
    
    unsigned long startMs = millis();
    while (millis() - startMs < sureMs) {
      server.handleClient(); // Acil durdurma veya hortum çekme HTTP isteklerini anında yakalar!
      ArduinoOTA.handle();   // OTA dinleyicisi
      if (emergencyStopActive) {
        Serial.println("[KESINTI] Acil durdurma tetiklendi, motor calisirken aninda durduruldu!");
        break;
      }
      delay(5); // 5ms hassas kontrol çözünürlüğü
    }
    
    digitalWrite(hedefPin, LOW);  
    Serial.println(">>> ISLEM TAMAMLANDI VEYA KESILDI! <<<\n");
  }
}

void loop() {
  ArduinoOTA.handle(); // Kablosuz güncelleme isteklerini dinle
  DateTime now = rtc.now(); 
  server.handleClient(); // Lokal HTTP İsteklerini Dinle

  // --- GÜVENLİK KİLİDİ (WATCHDOG TIMER) ---
  // Hortum doldurma sırasında Wi-Fi koparsa veya 1.5 sn sinyal gelmezse motoru anında kapat!
  for (int p = 1; p <= 4; p++) {
    if (pumpPrimingActive[p]) {
      if (millis() - primeLastHeartbeat[p] > 1500) {
        int pin = (p == 1) ? POMPA_1 : (p == 2) ? POMPA_2 : (p == 3) ? POMPA_3 : POMPA_4;
        digitalWrite(pin, LOW);
        pumpPrimingActive[p] = false;
        Serial.print("[GUVENLIK] Sinyal koptugu icin "); Serial.print(p); Serial.println(". Pompa kapatildi!");
      }
    }
  } 

  // --- 1. AŞAMA: GÜBRE MOTORLARI & KOMUT SENKRONİZASYONU (Her 10 Saniyede Bir) ---
  if (millis() - sonİnternetKontrol > 10000) {
    sonİnternetKontrol = millis();
    
    if (WiFi.status() == WL_CONNECTED) {
      kalpatisiGonder(); // Canlılık sinyalini Supabase'e gönder

      HTTPClient http;
      String endpoint = supabaseUrl + "/rest/v1/schedules?select=*&is_active=eq.true";
      http.begin(endpoint);
      http.addHeader("apikey", supabaseKey);
      http.addHeader("Authorization", "Bearer " + supabaseKey);
      
      int httpResponseCode = http.GET();
      if (httpResponseCode == 200) {
        String payload = http.getString();
        DynamicJsonDocument doc(4096);
        DeserializationError error = deserializeJson(doc, payload);
        
        if (!error) {
          int geciciGorevSayisi = 0;
          HafizaGorev geciciHafiza[20];

          for (JsonObject item : doc.as<JsonArray>()) {
            int id = item["id"];
            int pump_id = item["pump_id"];
            
            // Hassas süre hesaplama: duration_ms varsa doğrudan al, yoksa duration_seconds (ondalıklı/tam) üzerinden milisaniyeye çevir
            unsigned long durationMs = 0;
            if (item.containsKey("duration_ms") && !item["duration_ms"].isNull()) {
              durationMs = item["duration_ms"].as<unsigned long>();
            } else if (item.containsKey("duration_seconds") && !item["duration_seconds"].isNull()) {
              float dSec = item["duration_seconds"].as<float>();
              durationMs = (unsigned long)(dSec * 1000.0f);
            }
            if (durationMs < 50) durationMs = 50; // Donanımsal alt sınır

            bool is_one_time = item["is_one_time"];
            
            // Eğer manuel anlık tetikleme ise anında yap ve veritabanından pasife çek!
            if (is_one_time == true) {
              Serial.print("\n[ANLIK GOREV] Arayuzden manuel tetikleme geldi! Sure: ");
              Serial.print(durationMs); Serial.println(" ms");
              motorCalistir(pump_id, durationMs);
              goreviPasifYap(id); 
            } 
            // Eğer programlı görev ise, saati geldiğinde yapmak üzere HAFIZAYA AL
            else {
              String run_time = item["run_time"].as<String>(); 
              geciciHafiza[geciciGorevSayisi].pump_id = pump_id;
              geciciHafiza[geciciGorevSayisi].hour = run_time.substring(0, 2).toInt();
              geciciHafiza[geciciGorevSayisi].minute = run_time.substring(3, 5).toInt();
              geciciHafiza[geciciGorevSayisi].duration_ms = durationMs;

              // Esnek Zamanlama Alanları
              String stype = item["schedule_type"] | "daily";
              strncpy(geciciHafiza[geciciGorevSayisi].schedule_type, stype.c_str(), sizeof(geciciHafiza[geciciGorevSayisi].schedule_type));

              for(int d = 1; d <= 7; d++) geciciHafiza[geciciGorevSayisi].days_of_week[d] = false;
              if (item.containsKey("days_of_week") && !item["days_of_week"].isNull()) {
                JsonArray days = item["days_of_week"];
                for(int d : days) {
                  if(d >= 1 && d <= 7) geciciHafiza[geciciGorevSayisi].days_of_week[d] = true;
                }
              } else {
                for(int d = 1; d <= 7; d++) geciciHafiza[geciciGorevSayisi].days_of_week[d] = true;
              }

              geciciHafiza[geciciGorevSayisi].interval_days = item["interval_days"] | 1;

              if (item.containsKey("start_date") && !item["start_date"].isNull()) {
                String sDate = item["start_date"].as<String>();
                int yr = sDate.substring(0, 4).toInt();
                int mo = sDate.substring(5, 7).toInt();
                int dy = sDate.substring(8, 10).toInt();
                DateTime dt(yr, mo, dy, 0, 0, 0);
                geciciHafiza[geciciGorevSayisi].start_timestamp = dt.unixtime();
              } else {
                geciciHafiza[geciciGorevSayisi].start_timestamp = now.unixtime();
              }

              geciciGorevSayisi++;
            }
          }
          
          // İnternetten çekilen programlı görevleri ESP32'nin ana hafızasına aktar
          hafizadakiGorevSayisi = geciciGorevSayisi;
          for(int i = 0; i < geciciGorevSayisi; i++) {
            gunlukProgramlar[i] = geciciHafiza[i];
          }
        }
      }
      http.end();
    } else {
      Serial.println("[UYARI] WiFi Baglantisi Koptu! Offline (Cevrimdisi) mod devrede.");
    }
  }

  // --- 2. AŞAMA: DS18B20 SU SICAKLIĞI KONTROLÜ (Her 5 Dakikada Bir / 300.000 ms) ---
  if (millis() - sonSicaklikKontrol > 300000 || sonSicaklikKontrol == 0) {
    sonSicaklikKontrol = millis();
    sicaklikGonder();
  }

  // --- 3. AŞAMA: TDS & EC SU KALİTESİ KONTROLÜ (Saat Başı / 3.600.000 ms) ---
  if (millis() - sonTdsKontrol > 3600000 || sonTdsKontrol == 0) {
    sonTdsKontrol = millis();
    suKalitesiGonder();
  }

  // --- 4. AŞAMA: DOZLAMA SONRASI 10 DAKİKALIK ETKİ TAKİBİ ---
  if (sonDozlama.aktif && (millis() - sonDozlama.baslangicMs > 600000)) {
    sonDozlama.aktif = false;
    float tdsAfter = 0.0;
    float ecAfter = 0.0;
    tdsEcOku(tdsAfter, ecAfter);
    float tdsDelta = tdsAfter - sonDozlama.tdsBefore;

    Serial.print("\n[DOZAJ ETKİSİ] 10 Dakika Doldu! TDS Öncesi: ");
    Serial.print(sonDozlama.tdsBefore, 1);
    Serial.print(" PPM | Sonrası: ");
    Serial.print(tdsAfter, 1);
    Serial.print(" PPM | Fark: ");
    Serial.print(tdsDelta, 1);
    Serial.println(" PPM");

    if (WiFi.status() == WL_CONNECTED) {
      HTTPClient http;
      String endpoint = supabaseUrl + "/rest/v1/dosing_logs?pump_id=eq." + String(sonDozlama.pumpId) + "&order=created_at.desc&limit=1";
      http.begin(endpoint);
      http.addHeader("apikey", supabaseKey);
      http.addHeader("Authorization", "Bearer " + supabaseKey);
      http.addHeader("Content-Type", "application/json");
      String body = "{\"tds_after\":" + String(tdsAfter, 1) + ",\"tds_delta\":" + String(tdsDelta, 1) + "}";
      http.PATCH(body);
      http.end();
    }
  }



  // --- 5. AŞAMA: OFFLINE HAFIZA KONTROLÜ VE MOTOR TETİKLEME (Her Saniye) ---
  if (millis() - sonGorevKontrol > 1000) {
    sonGorevKontrol = millis();
    
    // ESP32 kendi hafızasındaki listeyi sürekli kontrol eder (İnternet olsa da olmasa da)
    for(int i = 0; i < hafizadakiGorevSayisi; i++) {
      int pId = gunlukProgramlar[i].pump_id;
      
      if(now.hour() == gunlukProgramlar[i].hour && now.minute() == gunlukProgramlar[i].minute) {
        // Bu motor bu dakika içinde henüz çalışmadıysa zamanlama kuralını denetle!
        if(sonCalisanDakika[pId] != now.minute()) {
          bool calismaliMi = false;
          String stype = String(gunlukProgramlar[i].schedule_type);

          if (stype == "weekly") {
            int bugunNo = now.dayOfTheWeek();
            if (bugunNo == 0) bugunNo = 7; // Pazar = 7 (1: Pzt ... 7: Pzr)
            if (gunlukProgramlar[i].days_of_week[bugunNo]) {
              calismaliMi = true;
            }
          } else if (stype == "interval") {
            int interval = gunlukProgramlar[i].interval_days;
            if (interval <= 1) {
              calismaliMi = true;
            } else {
              uint32_t simdiUnix = now.unixtime();
              uint32_t baslangicUnix = gunlukProgramlar[i].start_timestamp;
              if (simdiUnix >= baslangicUnix) {
                long gunFarki = (simdiUnix - baslangicUnix) / 86400;
                if (gunFarki % interval == 0) {
                  calismaliMi = true;
                }
              }
            }
          } else {
            // "daily"
            calismaliMi = true;
          }

          if (calismaliMi) {
            Serial.println("\n[OTOMATIK PROGRAM] Hafizadaki gorev calistiriliyor!");
            motorCalistir(pId, gunlukProgramlar[i].duration_ms);
            logEkle(pId, gunlukProgramlar[i].duration_ms); // Otomatik çalıştırma log kaydını Supabase'e ekle
            sonCalisanDakika[pId] = now.minute();
          }
        }
      }
    }
  }
}
